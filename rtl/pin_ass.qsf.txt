set_global_assignment -name DEVICE AUTO
set_global_assignment -name TOP_LEVEL_ENTITY fpgatemp_wrapper

set_global_assignment -name LAST_QUARTUS_VERSION "25.1std.0 Lite Edition"
set_global_assignment -name PROJECT_OUTPUT_DIRECTORY output_files
set_global_assignment -name EDA_SIMULATION_TOOL "Questa Altera FPGA (Verilog)"
set_global_assignment -name VERILOG_FILE mcp3008_spi_master.v
set_global_assignment -name VERILOG_FILE esp32_spi_follower.v
set_global_assignment -name VERILOG_FILE fpgatemp_wrapper.v

# ============================================================================
# PIN ASSIGNMENTS (DE1-SoC)
# ============================================================================

# Clock
set_location_assignment PIN_AF14 -to CLOCK_50

# Reset pushbutton (KEY0)
set_location_assignment PIN_AA14 -to KEY[0]

# MCP3008 side - FPGA is SPI master (GPIO_0[0:3])
set_location_assignment PIN_AC18 -to adc_sclk
set_location_assignment PIN_Y17  -to adc_cs_n
set_location_assignment PIN_AD17 -to adc_mosi
set_location_assignment PIN_Y18  -to adc_miso

# ESP32 side - FPGA is SPI slave (GPIO_0[4:7])
set_location_assignment PIN_AK16 -to esp_sclk
set_location_assignment PIN_AK18 -to esp_cs_n
set_location_assignment PIN_AK19 -to esp_mosi
set_location_assignment PIN_AJ19 -to esp_miso

# IO Standards
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to CLOCK_50
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to KEY[0]
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to adc_sclk
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to adc_cs_n
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to adc_mosi
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to adc_miso
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to esp_sclk
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to esp_cs_n
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to esp_mosi
set_instance_assignment -name IO_STANDARD "3.3-V LVTTL" -to esp_miso
